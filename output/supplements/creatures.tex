\begin{DndMonster}[float=t]{Draconic Spirit}
\DndMonsterType{Large dragon, Unaligned}
\DndMonsterBasics[
armor-class = {14 + the level of the spell (natural armor)},
hit-points = {50 + 10 for each spell level above 5th (the dragon has a number of Hit Dice [d10s] equal to the level of the spell)},
speed = {30 ft., fly 60 ft., swim 30 ft.}
]
\DndMonsterAbilityScores[
 str = 19,
 dex = 14,
 con = 17,
 int = 10,
 wis = 14,
 cha = 14
]
\DndMonsterDetails[
damage-resistances = {acid, cold, fire, lightning, poison (Chromatic and Metallic Only); force, necrotic, psychic, radiant, thunder (Gem Only)},
condition-immunities = {charmed, frightened, poisoned},
senses = {blindsight 30 ft., darkvision 60 ft., passive perception 12},
languages = {Draconic, understands the languages you speak},
]
\DndMonsterAction{Shared Resistances}
When you summon the dragon, choose one of its damage resistances. You have resistance to the chosen damage type until the spell ends.

\DndMonsterSection{Actions}
\DndMonsterAction{Multiattack}
The dragon makes a number of Rend attacks equal to half the spell's level (rounded down), and it uses Breath Weapon.

\DndMonsterAction{Rend}
\textsl{Melee Weapon Attack:} your spell attack modifier to hit, reach 10 ft., one target. \textsl{Hit:} 1d6 + 4 + spell level piercing damage.

\DndMonsterAction{Breath Weapon}
The dragon exhales destructive energy in a 30-foot cone. Each creature in that area must make a Dexterity saving throw against your spell save DC. A creature takes 2d6 damage of a type this dragon has resistance to (your choice) on a failed save, or half as much damage on a successful one.

\end{DndMonster}
